<script>
(function(){
  var root=document.querySelector('[data-dv2-slider]');
  if(!root)return;
  var slides=[].slice.call(root.querySelectorAll('[data-dv-slide]'));
  var dots=[].slice.call(root.querySelectorAll('[data-dv-dot]'));
  var reduced=window.matchMedia&&window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var current=0,timer;
  function load(i){var img=slides[i].querySelector('img[data-src]');if(img){img.src=img.getAttribute('data-src');img.removeAttribute('data-src')}}
  function show(i){current=i;load(i);slides.forEach(function(s,n){var active=n===i;s.classList.toggle('is-active',active);s.setAttribute('aria-hidden',active?'false':'true')});dots.forEach(function(d,n){var active=n===i;d.classList.toggle('is-active',active);if(active)d.setAttribute('aria-current','true');else d.removeAttribute('aria-current')})}
  function start(){if(reduced)return;clearInterval(timer);timer=setInterval(function(){show((current+1)%slides.length)},5000)}
  dots.forEach(function(d){d.addEventListener('click',function(){show(Number(d.getAttribute('data-dv-dot')));start()})});
  if(!reduced){window.setTimeout(function(){for(var i=1;i<slides.length;i++)load(i)},1200);start()}
  document.addEventListener('visibilitychange',function(){if(document.hidden)clearInterval(timer);else start()});
})();
</script>
