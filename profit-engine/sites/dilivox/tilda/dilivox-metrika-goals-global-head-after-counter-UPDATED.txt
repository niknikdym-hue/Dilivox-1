  function getScrollPercentForPage() {
    var doc = document.documentElement;
    var body = document.body;

    var scrollTop = window.pageYOffset || doc.scrollTop || body.scrollTop || 0;
    var scrollHeight = Math.max(
      body.scrollHeight,
      doc.scrollHeight,
      body.offsetHeight,
      doc.offsetHeight,
      body.clientHeight,
      doc.clientHeight
    );

    var windowHeight = window.innerHeight || doc.clientHeight;
    var scrollable = scrollHeight - windowHeight;

    if (scrollable <= 0) return 100;

    return Math.round((scrollTop / scrollable) * 100);
  }

  function getReadingPercent(article) {
    if (!article) return 0;

    var rect = article.getBoundingClientRect();
    var articleTop = rect.top + window.pageYOffset;
    var articleHeight = article.offsetHeight;
    var windowHeight = window.innerHeight || document.documentElement.clientHeight;
    var scrollTop = window.pageYOffset || document.documentElement.scrollTop || 0;
    var readPosition = scrollTop + windowHeight - articleTop;
    var percent = Math.round((readPosition / articleHeight) * 100);

    if (percent < 0) return 0;
    if (percent > 100) return 100;

    return percent;
  }

  function watchElement(selector, goalId, threshold) {
    var element = document.querySelector(selector);
    if (!element || !('IntersectionObserver' in window)) return;

    var observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          sendGoal(goalId);
          observer.disconnect();
        }
      });
    }, {
      threshold: threshold || 0.35
    });

    observer.observe(element);
  }

  function bindClick(selector, goalId, additionalGoalId) {
    var elements = document.querySelectorAll(selector);
    if (!elements.length) return;

    elements.forEach(function (element) {
      if (element.dataset.dvGoalBound === goalId) return;
      element.dataset.dvGoalBound = goalId;

      element.addEventListener('click', function () {
        sendGoal(goalId);
        if (additionalGoalId) sendGoal(additionalGoalId);
      });
    });
  }

  function getStoryProgress(slug) {
    var raw = parseInt(storageGet(PROGRESS_PREFIX + slug, '0'), 10);
    if (!Number.isFinite(raw)) return 0;
    if (raw < 0) return 0;
    if (raw > 100) return 100;
    return raw;
  }

  function setStoryProgress(slug, percent) {
    if (!slug) return;

    var status = getStoryStatus(slug);
    var current = getStoryProgress(slug);
    var rounded = Math.round(percent || 0);

    if (status !== 'read' && current >= 100) current = 95;
    if (status !== 'read' && rounded >= 100) rounded = 95;

    var next = Math.max(current, rounded);

    if (next > 100) next = 100;
    storageSet(PROGRESS_PREFIX + slug, String(next));
  }

  }

  function ensureCatalogProgressBadge(card, statusBadge) {
    var progressBadge = card.querySelector('[data-dv-story-progress]');

    if (!progressBadge) {
      var meta = card.querySelector('.dv-story-card__meta');
      progressBadge = document.createElement('span');
      progressBadge.className = 'dv-story-card__progress';
      progressBadge.setAttribute('data-dv-story-progress', '');
      progressBadge.hidden = true;

      if (meta) {
        meta.appendChild(progressBadge);
      } else {
        statusBadge.insertAdjacentElement('afterend', progressBadge);
      }
    }

    return progressBadge;
  }

  function renderCatalogProgress(progressBadge, status, progress) {
    if (!progressBadge) return;

    progressBadge.textContent = '';
    progressBadge.classList.remove('is-started', 'is-read');

    if (status !== 'started' && status !== 'read') {
      progressBadge.hidden = true;
      progressBadge.removeAttribute('aria-label');
      progressBadge.style.removeProperty('--dv-card-progress');
      return;
    }

    var fillPercent = status === 'read'
      ? 100
      : Math.min(Math.max(progress || 18, 18), 78);

    var label = document.createElement('span');
    label.className = 'dv-story-card__progress-label';
    label.textContent = status === 'read' ? 'готово' : 'в процессе';

    var track = document.createElement('span');
    track.className = 'dv-story-card__progress-track';
    track.setAttribute('aria-hidden', 'true');

    var fill = document.createElement('span');
    fill.className = 'dv-story-card__progress-fill';

    track.appendChild(fill);
    progressBadge.appendChild(label);
    progressBadge.appendChild(track);

    progressBadge.classList.add(status === 'read' ? 'is-read' : 'is-started');
    progressBadge.style.setProperty('--dv-card-progress', fillPercent + '%');
    progressBadge.setAttribute(
      'aria-label',
      status === 'read' ? 'История прочитана' : 'История начата'
    );
    progressBadge.hidden = false;
  }

  function applyCatalogStatuses() {
    var badges = document.querySelectorAll('[data-dv-story-status]');

    badges.forEach(function (badge) {
      var slug = badge.getAttribute('data-dv-story-status');
      var status = getStoryStatus(slug);
      var progress = getStoryProgress(slug);
      var card = badge.closest('.dv-story-card');
      var progressBadge = card ? ensureCatalogProgressBadge(card, badge) : null;

      badge.textContent = statusText[status];
      badge.classList.remove('is-new', 'is-started', 'is-read');
      badge.classList.add(statusClass[status]);

      renderCatalogProgress(progressBadge, status, progress);
    });
  }

  function getCatalogRoot() {
    return document.querySelector('[data-dv-page="stories-list"], .dv-stories-list');
  }

  function normalizeStoriesFilter(mode) {
    var value = mode || 'all';
    var aliases = {
      detective: 'detektivy',
      detektivnye: 'detektivy',
      fantasy: 'fantastika',
      fantasticheskie: 'fantastika',
      thriller: 'trillery',
      triller: 'trillery',
      trillery: 'trillery',
      horror: 'uzhasy',
      uzhas: 'uzhasy',
      uzhasy: 'uzhasy',
      adventure: 'priklyucheniya',
      adventures: 'priklyucheniya',
      priklyuchenie: 'priklyucheniya',
      priklyucheniya: 'priklyucheniya'
    };

    return aliases[value] || value;
  }

  function normalizeStoryFormat(mode) {
    var value = mode || 'all';
    return value === 'all' || storyFormats[value] ? value : 'all';
  }

  function getRequestedStoriesFilter() {
    try {
      return normalizeStoriesFilter(new URLSearchParams(location.search).get('filter') || 'all');
    } catch (e) {
      return 'all';
    }
  }

  function getActiveStoriesFilter() {
    var root = getCatalogRoot();
    if (!root) return 'all';

    var active = root.querySelector('[data-dv-filter-control][aria-pressed="true"]');
    return active ? normalizeStoriesFilter(active.getAttribute('data-dv-filter') || 'all') : 'all';
  }

  function getRequestedStoriesFormat() {
    try {
      return normalizeStoryFormat(new URLSearchParams(location.search).get('format') || 'all');
    } catch (e) {
      return 'all';
    }
  }

  function getActiveStoryFormatFilter() {
    var root = getCatalogRoot();
    if (!root) return 'all';

    var active = root.querySelector('[data-dv-format-control][aria-pressed="true"]');
    return active ? normalizeStoryFormat(active.getAttribute('data-dv-format-filter') || 'all') : 'all';
  }

  function getStoryCardGenre(card) {
    return normalizeStoriesFilter(
      card.getAttribute('data-dv-genre') ||
      card.getAttribute('data-dv-category') ||
      'detektivy'
    );
  }

  function getStoryCardFormat(card) {
    return normalizeStoryFormat(card.getAttribute('data-dv-format') || 'short');
  }

  function hasMultipleStoryFormats(root) {
    var formats = {};
    Array.prototype.slice.call(root.querySelectorAll('.dv-story-card')).forEach(function (card) {
      formats[getStoryCardFormat(card)] = true;
    });

    return Object.keys(formats).length > 1;
  }

  function syncStoryFormatFilterVisibility(root) {
    var formatTools = Array.prototype.slice.call(root.querySelectorAll('[data-dv-format-tools]'));
    if (!formatTools.length) return;

    var shouldShow = hasMultipleStoryFormats(root);
    formatTools.forEach(function (tool) {
      tool.hidden = !shouldShow;
    });
  }

  function applyStoriesCatalogFilters(genreMode, formatMode) {
    var root = getCatalogRoot();
    if (!root) return;

    var cards = Array.prototype.slice.call(root.querySelectorAll('.dv-story-card'));
    var activeGenre = normalizeStoriesFilter(genreMode || 'all');
    var activeFormat = normalizeStoryFormat(formatMode || getActiveStoryFormatFilter());

    cards.forEach(function (card) {
      var genre = getStoryCardGenre(card);
      var format = getStoryCardFormat(card);
      var genreMatches = activeGenre === 'all' || genre === activeGenre;
      var formatMatches = activeFormat === 'all' || format === activeFormat;
      card.hidden = !genreMatches || !formatMatches;
    });

  }

  function applyStoriesCategoryFilter(mode) {
    applyStoriesCatalogFilters(mode || getActiveStoriesFilter(), getActiveStoryFormatFilter());
  }

  function sortStoriesCatalog(mode) {
    var root = getCatalogRoot();
    if (!root) return;

    var grid = root.querySelector('[data-dv-stories-grid], .dv-stories__grid');
    if (!grid) return;

    var cards = Array.prototype.slice.call(grid.querySelectorAll('.dv-story-card'));
    var attribute = mode === 'popular' ? 'data-dv-popular' : 'data-dv-order';

    cards.sort(function (first, second) {
      var firstValue = parseFloat(first.getAttribute(attribute) || '0') || 0;
      var secondValue = parseFloat(second.getAttribute(attribute) || '0') || 0;
      return secondValue - firstValue;
    });

    if (mode === 'new' && getActiveStoriesFilter() === 'all') {
      var fixedSlugs = ['ona-vyshla-na-stsenu-dvazhdy', 'pansionat-tikhiy-uzhas'];
      var fixedCards = fixedSlugs.map(function (slug) {
        return cards.find(function (card) {
          return card.getAttribute('data-dv-story-slug') === slug;
        });
      }).filter(Boolean);
      var usedCards = fixedCards.slice();
      var usedGenres = {};

      fixedCards.forEach(function (card) {
        usedGenres[getStoryCardGenre(card)] = true;
      });

      var genreHighlights = [];
      cards.forEach(function (card) {
        var genre = getStoryCardGenre(card);
        if (genreHighlights.length >= 6 || usedCards.indexOf(card) !== -1 || usedGenres[genre]) return;
        genreHighlights.push(card);
        usedCards.push(card);
        usedGenres[genre] = true;
      });

      cards = fixedCards.concat(genreHighlights, cards.filter(function (card) {
        return usedCards.indexOf(card) === -1;
      }));
    }

    cards.forEach(function (card) {
      grid.appendChild(card);
    });

    applyStoriesCategoryFilter(getActiveStoriesFilter());
  }

  function initStoriesListFilters() {
    var root = getCatalogRoot();
    if (!root) return;

    syncStoryFormatFilterVisibility(root);

    var controls = Array.prototype.slice.call(root.querySelectorAll('[data-dv-filter-control]'));
    if (!controls.length) return;

    var requestedFilter = getRequestedStoriesFilter();
    var requestedControl = controls.find(function (control) {
      return normalizeStoriesFilter(control.getAttribute('data-dv-filter') || 'all') === requestedFilter;
    });

    if (requestedControl) {
      controls.forEach(function (item) {
        var isActive = item === requestedControl;
        item.classList.toggle('is-active', isActive);
        item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
      });
    }

    controls.forEach(function (control) {
      if (control.dataset.dvFilterBound === 'true') return;
      control.dataset.dvFilterBound = 'true';

      control.addEventListener('click', function () {
        var mode = normalizeStoriesFilter(control.getAttribute('data-dv-filter') || 'all');

        controls.forEach(function (item) {
          var isActive = item === control;
          item.classList.toggle('is-active', isActive);
          item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
        });

        applyStoriesCategoryFilter(mode);
        applyCatalogStatuses();
        sendGoal('dv_stories_filter_' + mode);
      });
    });

    applyStoriesCategoryFilter(getActiveStoriesFilter());
  }

  function initStoriesFormatFilters() {
    var root = getCatalogRoot();
    if (!root) return;

    syncStoryFormatFilterVisibility(root);

    var controls = Array.prototype.slice.call(root.querySelectorAll('[data-dv-format-control]'));
    if (!controls.length) return;

    var requestedFormat = getRequestedStoriesFormat();
    var requestedControl = controls.find(function (control) {
      return normalizeStoryFormat(control.getAttribute('data-dv-format-filter') || 'all') === requestedFormat;
    });

    if (requestedControl) {
      controls.forEach(function (item) {
        var isActive = item === requestedControl;
        item.classList.toggle('is-active', isActive);
        item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
      });
    }

    controls.forEach(function (control) {
      if (control.dataset.dvFormatBound === 'true') return;
      control.dataset.dvFormatBound = 'true';

      control.addEventListener('click', function () {
        var mode = normalizeStoryFormat(control.getAttribute('data-dv-format-filter') || 'all');

        controls.forEach(function (item) {
          var isActive = item === control;
          item.classList.toggle('is-active', isActive);
          item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
        });

        applyStoriesCatalogFilters(getActiveStoriesFilter(), mode);
        applyCatalogStatuses();
        sendGoal('dv_stories_format_' + mode);
      });
    });

    applyStoriesCatalogFilters(getActiveStoriesFilter(), getActiveStoryFormatFilter());
  }

  function initStoriesFilterJumps() {
    var root = getCatalogRoot();
    if (!root) return;

    Array.prototype.slice.call(root.querySelectorAll('[data-dv-filter-jump]')).forEach(function (button) {
      if (button.dataset.dvFilterJumpBound === 'true') return;
      button.dataset.dvFilterJumpBound = 'true';

      button.addEventListener('click', function () {
        var filter = normalizeStoriesFilter(button.getAttribute('data-dv-filter-jump') || 'all');
        var control = root.querySelector('[data-dv-filter-control][data-dv-filter="' + filter + '"]');

        if (control) {
          control.click();
          var catalog = root.querySelector('[data-dv-stories-grid], .dv-stories__grid');
          if (catalog && catalog.scrollIntoView) {
            catalog.scrollIntoView({ behavior: 'smooth', block: 'start' });
          }
        }
      });
    });
  }

  function initStoriesListControls() {
    var root = getCatalogRoot();
    if (!root) return;

    var controls = Array.prototype.slice.call(root.querySelectorAll('[data-dv-sort-control]'));
    if (!controls.length) return;

    controls.forEach(function (control) {
      if (control.dataset.dvSortBound === 'true') return;
      control.dataset.dvSortBound = 'true';

      control.addEventListener('click', function () {
        var mode = control.getAttribute('data-dv-sort') || 'new';

        controls.forEach(function (item) {
          var isActive = item === control;
          item.classList.toggle('is-active', isActive);
          item.setAttribute('aria-pressed', isActive ? 'true' : 'false');
        });

        sortStoriesCatalog(mode);
        applyCatalogStatuses();
      });
    });

    var active = controls.find(function (control) {
      return control.getAttribute('aria-pressed') === 'true';
    }) || controls[0];

    if (active) {
      sortStoriesCatalog(active.getAttribute('data-dv-sort') || 'new');
    }
  }

  function ensureStoryImage(root, slug) {
    var asset = storyAssets[slug];
    if (!asset || root.querySelector('[data-dv-story-image]')) return;

    var alt = 'Иллюстрация к истории «' + asset.title + '»';
    var hero = root.querySelector('.dv-story__hero');

    if (hero) {
      var section = document.createElement('section');
      section.className = 'dv-story__image-section';
      section.setAttribute('data-dv-story-image', '');
      section.setAttribute('aria-label', alt);

      var shell = document.createElement('div');
      shell.className = 'dv-story__shell dv-story__image-shell';

      var figure = document.createElement('figure');
      figure.className = 'dv-story__image-card';

      var image = document.createElement('img');
      image.src = asset.image;
      image.alt = alt;
      image.loading = 'lazy';

      figure.appendChild(image);
      shell.appendChild(figure);
      section.appendChild(shell);
      hero.insertAdjacentElement('afterend', section);
      return;
    }

    var wrap = root.querySelector('.dv-story__wrap') || root;
    var anchor = wrap.querySelector('.dv-story__meta') || wrap.querySelector('.dv-story__lead');
    if (!anchor) return;

    var fallbackFigure = document.createElement('figure');
    fallbackFigure.className = 'dv-story__image-card';
    fallbackFigure.setAttribute('data-dv-story-image', '');
    fallbackFigure.setAttribute('aria-label', alt);

    var fallbackImage = document.createElement('img');
    fallbackImage.src = asset.image;
    fallbackImage.alt = alt;
    fallbackImage.loading = 'lazy';

    fallbackFigure.appendChild(fallbackImage);
    anchor.insertAdjacentElement('afterend', fallbackFigure);
  }

  function initStoriesListGoals() {
    sendGoal('dv_stories_list_open');

    bindClick(
      '[data-dv-goal="story-card"], .dv-story-card, .dv-story-link',
      'dv_story_card_click'
    );
  }

  function createReadingProgressBar() {
    var existing = document.querySelector('[data-dv-reading-progress]');
    if (existing) return existing.querySelector('.dv-reading-progress__bar');

    var progress = document.createElement('div');
    progress.className = 'dv-reading-progress';
    progress.setAttribute('data-dv-reading-progress', '');
    progress.setAttribute('aria-hidden', 'true');

    var bar = document.createElement('span');
    bar.className = 'dv-reading-progress__bar';
    progress.appendChild(bar);
    document.body.appendChild(progress);

    return bar;
  }

  function revealStoryAdsAfterFinal(root) {
    Array.prototype.slice.call(root.querySelectorAll('[data-dv-ad-after-reveal]')).forEach(function (ad) {
      ad.hidden = false;
      ad.setAttribute('aria-hidden', 'false');
    });
  }

  function revealStoryChoice(root, slug, button, options) {
    options = options || {};

    var buttons = Array.prototype.slice.call(root.querySelectorAll('[data-dv-choice]'));
    var result = root.querySelector('[data-dv-choice-result]');
    var finalBlock = root.querySelector('[data-dv-final], [data-dv-reveal]');
    var proofBlock = root.querySelector('[data-dv-proof], [data-dv-facts]');

    if (!buttons.length || !button || !result || !finalBlock || !proofBlock) return;

    var isCorrect = button.getAttribute('data-dv-correct') === 'true';
    var selectedIndex = buttons.indexOf(button);

    buttons.forEach(function (item) {
      item.disabled = true;
      item.setAttribute('aria-disabled', 'true');
      item.setAttribute('aria-pressed', item === button ? 'true' : 'false');
      item.classList.remove('is-selected', 'is-correct', 'is-wrong');

      if (item.getAttribute('data-dv-correct') === 'true') {
        item.classList.add('is-correct');
      }
    });

    button.classList.add('is-selected');

    if (!isCorrect) {
      button.classList.add('is-wrong');
    }

    result.hidden = false;
    result.setAttribute('role', 'status');
    result.textContent = isCorrect
      ? 'Версия выдержала проверку. Теперь смотрим, какая деталь решила дело.'
      : 'Версия звучала убедительно, но решающая деталь ведёт в другую сторону.';

    finalBlock.hidden = false;
    proofBlock.hidden = false;
    finalBlock.setAttribute('aria-hidden', 'false');
    proofBlock.setAttribute('aria-hidden', 'false');
    finalBlock.classList.add('dv-is-revealed');
    proofBlock.classList.add('dv-is-revealed');
    revealStoryAdsAfterFinal(root);

    setStoryStatus(slug, 'read');
    setStoryProgress(slug, 100);

    if (!options.restore) {
      saveChoice(slug, {
        index: selectedIndex,
        correct: isCorrect,
        at: Date.now()
      });
      sendGoal('pe_version_selected');
      sendGoal('pe_story_completed');
    }

    if (options.scroll) {
      setTimeout(function () {
        var reducedMotion = window.matchMedia &&
          window.matchMedia('(prefers-reduced-motion: reduce)').matches;

        finalBlock.scrollIntoView({
          behavior: reducedMotion ? 'auto' : 'smooth',
          block: 'start'
        });
      }, 180);
    }
  }

  function initStoryChoice(root, slug) {
    var buttons = Array.prototype.slice.call(root.querySelectorAll('[data-dv-choice]'));
    if (!buttons.length) return;

    buttons.forEach(function (button, index) {
      button.setAttribute('aria-pressed', 'false');

      if (button.dataset.dvChoiceBound === 'true') return;
      button.dataset.dvChoiceBound = 'true';

      button.addEventListener('click', function () {
        revealStoryChoice(root, slug, button, {
          scroll: true
        });
      });
    });

    var savedChoice = getSavedChoice(slug);
    var status = getStoryStatus(slug);

    if (savedChoice && typeof savedChoice.index === 'number' && buttons[savedChoice.index]) {
      revealStoryChoice(root, slug, buttons[savedChoice.index], {
        restore: true,
        scroll: false
      });
    } else if (status === 'read') {
      var correctButton = buttons.find(function (button) {
        return button.getAttribute('data-dv-correct') === 'true';
      });

      if (correctButton) {
        revealStoryChoice(root, slug, correctButton, {
          restore: true,
          scroll: false
        });
      }
    }
  }

  function initStoryGoalsAndProgress(root, slug) {
    var article = root.querySelector('[data-dv-story-text], .dv-story-text');
    var progressBar = createReadingProgressBar();

    sendGoal('dv_story_open');

    setTimeout(function () {
      sendGoal('dv_story_time_60');
    }, 60000);

    setTimeout(function () {
      sendGoal('dv_story_time_180');
    }, 180000);

    function checkReadingGoals() {
      if (!article) return;

      var percent = getReadingPercent(article);
      var rounded = Math.round(percent);

      if (progressBar) {
        progressBar.style.width = rounded + '%';
      }

      root.style.setProperty('--dv-reading-progress', rounded + '%');
      setStoryProgress(slug, rounded);

      if (percent >= 10) sendGoal('dv_story_read_start');
      if (percent >= 25) sendGoal('dv_story_read_25');
      if (percent >= 50) sendGoal('dv_story_read_50');
      if (percent >= 75) {
        sendGoal('dv_story_read_75');
        sendGoal('pe_story_progress_75');
      }
      if (percent >= 90) sendGoal('dv_story_read_90');
      if (percent >= 98) sendGoal('dv_story_read_complete');
    }

    window.addEventListener('scroll', checkReadingGoals, { passive: true });
    window.addEventListener('resize', checkReadingGoals);

    checkReadingGoals();

    bindClick(
      '[data-dv-goal="next-story"], .dv-next-story',
      'dv_next_story_click',
      'pe_next_story_clicked'
    );

    bindClick(
      '[data-dv-goal="prev-story"], .dv-prev-story',
      'dv_prev_story_click'
    );

    bindClick(
      '[data-dv-goal="back-to-stories"], .dv-back-to-stories',
      'dv_back_to_stories_click'
    );
  }

  function initStoryPage() {
    var root = getStoryRoot();
    if (!root) return;

    var slug = getStorySlug(root);
    if (!slug) return;

    setStoryStatus(slug, 'started');
    ensureStoryImage(root, slug);
    initStoryChoice(root, slug);
    initStoryGoalsAndProgress(root, slug);
  }

  onReady(function () {
    initReturnVisitGoal();

    if (isHomePage()) {
      initHomeGoals();
    }

    if (isStoriesListPage()) {
      initStoriesListFilters();
      initStoriesFormatFilters();
      initStoriesListControls();
      initStoriesFilterJumps();
      applyCatalogStatuses();
      initStoriesListGoals();
    }

    if (isStoryPage()) {
      initStoryPage();
    }
  });
})();
